100 задач на конкурентность в Go
================================

1. Нужно считать из канала in ровно n чисел и вернуть их в порядке получения, не дожидаясь закрытия канала. Остальные значения должен получить другой участник, поэтому считывать их нельзя. n >= 0, отправитель пришлет не меньше n чисел; при n = 0 чтений быть не должно.
func ReceiveN(in <-chan int, n int) []int

2. Нужно считать числа из канала in до его закрытия отправителем и вернуть их сумму. Если чисел не было, нужно вернуть 0. Сумма помещается в int.
func SumStream(in <-chan int) int

3. Нужно считать числа из канала in до его закрытия отправителем и найти наименьшее. Нужно вернуть найденное число и true; если чисел не было - 0, false.
func MinStream(in <-chan int) (int, bool)

4. Нужно считать все числа из канала in до его закрытия отправителем. Если чисел ровно два, нужно вернуть их в порядке получения; иначе - 0, 0 и ошибку.
func ReadExactlyTwo(in <-chan int) (int, int, error)

5. Нужно вернуть уже заполненный и закрытый канал с числами nums в исходном порядке. Функция должна завершиться без запуска горутин и ожидания читателя; пустой nums дает ненулевой закрытый канал. Исходный слайс менять нельзя.
func PrepareNumbers(nums []int) <-chan int

6. Нужно запустить горутину, которая отправляет числа nums в канал в исходном порядке, затем закрывает его. Сразу нужно вернуть этот канал; его емкость равна capacity. capacity >= 0, получатель прочитает все значения до закрытия; nums никто не меняет до завершения отправки.
func BufferedSource(nums []int, capacity int) <-chan int

7. Нужно отправить числа nums в out по порядку и вернуться после последней отправки. out открыт и остается у вызывающей стороны - нельзя закрывать его; получатель работает параллельно. nums нельзя менять.
func SendAll(nums []int, out chan<- int)

8. Нужно считывать и отбрасывать числа из канала in до его закрытия отправителем. Сохранять значения и запускать дополнительную горутину нельзя.
func Drain(in <-chan int)

9. Нужно сохранить первые k чисел из канала in, затем считать и отбросить остальные до закрытия канала отправителем. Если чисел меньше k, нужно вернуть все. k >= 0; при k = 0 тоже нужно дочитать канал.
func FirstK(in <-chan int, k int) []int

10. Нужно параллельно посчитать вхождения target в left и right, дождаться обеих работ и вернуть количества в порядке аргументов. Для пустого слайса нужно вернуть 0. Входные данные менять нельзя.
func CountInBoth(left, right []int, target int) (int, int)

11. Нужно запустить переданные функции-проверки first и second в двух горутинах и дождаться обеих. Ошибка одной не отменяет вторую; нужно вернуть ошибки в порядке first, second.
func RunChecks(first, second func() error) (error, error)

12. Нужно запустить переданную функцию compute в отдельной горутине и сразу вернуть канал для результата. Когда вычисление закончится, нужно отправить полученное число в канал и закрыть его, не ожидая читателя.
func StartValue(compute func() int) <-chan int

13. Нужно запустить горутину, которая читает числа из in и для каждого по очереди вызывает переданную функцию consume. Сразу нужно вернуть канал-сигнал и закрыть его после закрытия in и завершения всех вызовов consume. Если чисел не было, сигнал тоже должен закрыться.
func StartConsumer(in <-chan int, consume func(int)) <-chan struct{}

14. Нужно выполнить work в фоне и сразу вернуть канал-сигнал без значений. Его закрытие после work должны увидеть все наблюдатели, включая начавших ждать после завершения.
func StartCompletion(work func()) <-chan struct{}

15. Нужно запустить background в новой горутине, а local выполнить в вызывающей горутине параллельно с ней. Нужно дождаться обеих и вернуть результаты в порядке local, background.
func RunAlongside(local, background func() int) (int, int) -- skipped

16. Нужно считывать строки из канала in до его закрытия отправителем и посчитать, сколько раз встретилась каждая. Регистр и пустая строка значимы; если строк не было, нужно вернуть пустую мапу.
func WordCounts(in <-chan string) map[string]int

17. Нужно считывать числа из канала in до его закрытия отправителем и проверить, что каждое следующее не меньше предыдущего. После нарушения тоже нужно дочитать канал. Если чисел нет или оно одно, нужно вернуть true.
func IsNonDecreasing(in <-chan int) bool

18. В одной фоновой горутине нужно последовательно вычислить next(0), ..., next(n-1) и отправить ответы в небуферизированный канал. Нужно вернуть канал сразу и закрыть после отправок; при n = 0 вызовов нет. n >= 0, получатель дочитает выход.
func Generate(n int, next func(int) int) <-chan int

19. Нужно дождаться закрытия всех каналов signals. Владельцы гарантированно закрывают их без отправки значений; каналы не nil, могут повторяться и быть уже закрыты. Пустой список не требует ожидания.
func AwaitSignals(signals []<-chan struct{})

20. Сразу нужно вернуть канал, а read вызвать в фоне только после закрытия ready. Нужно отправить один ответ и закрыть выход без ожидания читателя. Владелец гарантированно закроет ready без отправки значений.
func ReadAfterReady(ready <-chan struct{}, read func() int) <-chan int

21. Нужно выполнить first и second параллельно, затем вызвать finish один раз. Сразу нужно вернуть канал-сигнал и закрыть его после завершения всех трех вызовов.
func RunPairThenFinish(first, second, finish func()) <-chan struct{}

22. Нужно запустить все переданные функции jobs параллельно и сразу вернуть общий канал их результатов в порядке получения. Емкость канала равна len(jobs): работы и закрытие канала должны завершиться без читателя. Не более 32 работ; пустой список дает закрытый канал.
func StartBatch(jobs []func() int) <-chan int

23. Нужно выполнить calculate в фоне и сразу вернуть канал одного ответа. При ошибке нужно отправить Value = 0 и исходный Err, при успехе сохранить значение. Отправка и закрытие выхода не должны ждать читателя.
type Outcome struct { Value int; Err error }
func StartCalculation(calculate func() (int, error)) <-chan Outcome

24. Нужно отправить value в out и дождаться закрытия processed, подтверждающего обработку. Получатель работает параллельно и закроет processed без отправки значений после обработки; out открыт и может иметь буфер. Переданные каналы нельзя закрывать.
func SendAndWait(value int, out chan<- int, processed <-chan struct{})

25. Нужно считать из канала in длину пакета и указанное число элементов; затем канал должен закрыться. Отсутствие заголовка, отрицательная длина, недостача или избыток элементов дают nil и ошибку; в любом случае нужно дочитать канал до закрытия отправителем. Неотрицательная длина не больше 1024; длина 0 задает пустой пакет.
func ReadFrame(in <-chan int) ([]int, error)

26. Нужно считать строки из in до закрытия канала отправителем и вернуть те, что получены после первого marker. Само первое совпадение нужно исключить, последующие marker сохранить. Если совпадений нет, нужно вернуть пустой результат.
func AfterMarker(in <-chan string, marker string) []string

27. Нужно считывать числа из каналов left и right и объединять их по позициям: первое с первым, второе со вторым. Нужно дочитать оба канала до закрытия отправителями; при разной длине нужно вернуть полные пары и ошибку, иначе - пары и nil. Каналы разные, отправители независимы; два канала без значений допустимы.
type NumberPair struct { Left, Right int }
func PairStreams(left, right <-chan int) ([]NumberPair, error)

28. Нужно считывать сообщения из in до закрытия канала отправителем и разбить их на разделы: End = true завершает текущий раздел, его Value игнорируется. Нужно сохранить пустые разделы; незавершенный раздел при закрытии дает nil и ошибку. Если сообщений нет, разделов тоже нет; слайсы результата не разделяют элементы.
type SectionItem struct { Value int; End bool }
func ReadSections(in <-chan SectionItem) ([][]int, error)

29. Нужно считать числа из in до закрытия канала отправителем и вернуть последние k в исходном порядке. Нужно использовать O(k) памяти вместе с результатом. k >= 0; при k = 0 тоже нужно дочитать канал и вернуть пустой слайс.
func LastK(in <-chan int, k int) []int

30. Нужно считывать числа из in и последовательно вызывать apply до первой ошибки. После нее больше нельзя вызывать apply, но нужно дочитать канал до закрытия отправителем. Нужно вернуть первую ошибку или nil.
func ApplyUntilError(in <-chan int, apply func(int) error) error

31. Параллельно нужно вызвать analyze для каждой части и вернуть ответы в порядке parts, дождавшись всех вызовов. Нельзя пропускать пустые и повторяющиеся части. Частей не более 32; данные никто не меняет, analyze только читает их.
func AnalyzeParts(parts [][][]int, analyze func([][]int) int) []int

32. Параллельно нужно найти первое вхождение target в двух соседних половинах nums. Нужно дождаться обеих работ и вернуть минимальный индекс во всем слайсе или -1, если числа нет. nums нельзя менять.
func FindFirst(nums []int, target int) int

33. Параллельно нужно посчитать частоты чисел в частях chunks и объединить результаты в одну мапу. Итоговую мапу изменяет один получатель; переданные ему локальные мапы отправители больше не используют. Частей не более 32, пустые тоже дают результат; chunks нельзя менять.
func CountChunks(chunks [][][]int) map[int]int

34. Параллельно нужно вызвать left и right с независимыми копиями nums и вернуть измененные копии в этом порядке после обоих вызовов. Исходный слайс нельзя менять; результаты не должны разделять элементы друг с другом или с ним. Обе функции вызываются и для пустого входа; они не сохраняют слайсы.
func ForkEdits(nums []int, left, right func([]int)) ([]int, []int)

35. Нужно запустить jobs параллельно и сразу вернуть небуферизированный канал ответов с индексами работ в порядке получения. Нужно передать ровно один ответ на работу и закрыть канал после всех работ. Работ не более 32; получатель прочитает все ответы до закрытия, нельзя накапливать их все перед первой отправкой.
type IndexedValue struct { Index, Value int }
func StartIndexed(jobs []func() int) <-chan IndexedValue

36. Нужно запустить проверки независимо: каждая ждет закрытия своего Ready, затем вызывает Check. Закрытый Ready поздней проверки не должен ждать разрешения ранней. Нужно дождаться всех и вернуть ошибки по индексам, включая nil; проверок не более 32, их владельцы гарантированно закрывают Ready без отправки значений.
type GatedCheck struct { Ready <-chan struct{}; Check func() error }
func RunReadyChecks(checks []GatedCheck) []error

37. Нужно параллельно вызвать left и right и дождаться обеих операций. При равных строках без ошибок нужно вернуть строку; при различии - ошибку. Ошибки источников имеют приоритет над сравнением, left - над right; при любой ошибке строковый результат пустой.
func CompareSources(left, right func() (string, error)) (string, error)

38. Нужно считать ответы из in до закрытия канала отправителем и сопоставить с уникальными requestIDs. Нужно вернуть ответы в порядке запросов. Неизвестный ID, повтор или отсутствие ответа дают nil и ошибку, но не отменяют чтение до закрытия. Для пустого списка запросов корректен только канал без ответов.
type Reply struct { ID string; Value int }
func MatchReplies(requestIDs []string, in <-chan Reply) ([]Reply, error)

39. Нужно считать n частей из канала in и объединить их Values в порядке Index от 0 до n-1. Пропуск, повтор или неверный индекс дают nil и ошибку; в любом случае нужно дочитать канал до закрытия отправителем. n >= 0, пустая часть учитывается; отправитель не меняет Values после передачи, результат не разделяет с ними элементы.
type ResultPart struct { Index int; Values []int }
func JoinParts(in <-chan ResultPart, n int) ([]int, error)

40. Нужно выполнить все checks параллельно и вернуть, набралось ли хотя бы required ответов true. Даже при уже известном результате нужно дождаться всех; required = 0 не отменяет вызовы. Проверок не более 32, 0 <= required <= len(checks).
func CheckQuorum(checks []func() bool, required int) bool

41. Нужно считывать строки из in до закрытия канала отправителем и последовательно вызывать apply для каждой, продолжая после ошибок. Нужно вернуть по одному error на строку, включая nil для успешных вызовов.
func ApplyAll(in <-chan string, apply func(string) error) []error

42. Нужно запустить горутину, которая читает числа из in и отправляет отрицательные в один небуферизированный канал, остальные - в другой. Сразу нужно вернуть эти два канала в указанном порядке; нужно сохранить порядок чисел в каждом и закрыть оба после закрытия in отправителем. Получатели одновременно читают оба канала до закрытия.
func RouteBySign(in <-chan int) (<-chan int, <-chan int)

43. Нужно считывать числа из in до закрытия канала отправителем. Для каждого последовательно нужно вызвать check, а при успехе - apply; нужно вернуть ошибку check либо результат apply на соответствующей позиции. Нельзя останавливать обработку следующих чисел после ошибок.
func CheckThenApply(in <-chan int, check, apply func(int) error) []error

44. В одной горутине нужно считывать числа из in и последовательно вызывать transform. Сразу нужно вернуть канал ответов: один на число, при ошибке Value = 0 и исходный Err. После закрытия in отправителем нужно закрыть ответы; нельзя накапливать их все перед отправкой, получатель читает до закрытия.
type Outcome struct { Value int; Err error }
func TransformStream(in <-chan int, transform func(int) (int, error)) <-chan Outcome

45. Нужно считать ответы из in до закрытия канала отправителем и разделить их на успешные значения и ошибки. Нужно сохранить порядок внутри каждой группы; при Err != nil нужно игнорировать Value, успешный ноль сохранить.
type Outcome struct { Value int; Err error }
func PartitionResults(in <-chan Outcome) ([]int, []error)

46. Нужно считать ошибки из in до закрытия канала отправителем и вернуть первую ненулевую в порядке получения или nil, если ее нет. Значение nil из канала не означает его закрытие.
func FirstError(in <-chan error) error

47. Нужно считать числа из каналов a и b до закрытия отправителями и вернуть отдельную сумму для каждого. Если это один канал, нужно считать его один раз и вернуть сумму дважды. У разных каналов независимые отправители; суммы помещаются в int.
func SumSources(a, b <-chan int) (int, int)

48. Параллельно нужно вызвать left и right и дождаться обеих. При успехе нужно вызвать build с их значениями в порядке источников; иначе build нельзя вызывать и нужно вернуть ошибку left, если она есть, иначе right. Любая ошибка, включая ошибку build, дает числовой результат 0.
func BuildFromTwo(left, right func() (int, error), build func(int, int) (int, error)) (int, error)

49. В фоне нужно получить ресурс через acquire, вызвать use и затем release, даже при ошибке use. Ошибка acquire отменяет остальные вызовы; при нескольких ошибках приоритет: acquire, use, release. Сразу нужно вернуть канал одного ответа и закрыть его после всех действий без ожидания читателя.
func StartResource(acquire func() (int, error), use, release func(int) error) <-chan error

50. В фоне нужно выполнить load и при успехе вызвать audit. Успешную загрузку нужно сохранить как Value и Loaded = true даже при ошибке audit; при ошибке load нужно вернуть Value = 0, Loaded = false и нельзя вызывать audit. Сразу нужно вернуть канал одного отчета и закрыть после всех вызовов без ожидания читателя.
type LoadReport struct { Value int; Loaded bool; LoadErr, AuditErr error }
func StartAuditedLoad(load func() (int, error), audit func(int) error) <-chan LoadReport

51. Нужно запустить горутину-владельца суммы initial и сразу вернуть небуферизированные каналы запросов и ответов. На каждое изменение она отвечает новым балансом. Вызывающая сторона чередует запрос и ответ, затем закрывает запросы и дочитывает ответы; без запросов ответов нет. Арифметика помещается в int.
func StartAccumulator(initial int) (chan<- int, <-chan int)

52. Нужно передать изменения одной горутине-владельцу баланса initial через последовательный запрос-ответ. Она отклоняет изменение, если баланс станет отрицательным, но продолжает обработку; нужно вернуть ответы в порядке deltas после завершения обработчика. initial >= 0, арифметика помещается в int; deltas нельзя менять.
type BalanceReply struct { Applied bool; Balance int }
func ApplyDeltas(initial int, deltas []int) []BalanceReply

53. Нужно считывать строки из in и последовательно вызывать apply. После каждого вызова нужно отправлять индекс строки в acknowledgements, начиная с 0. Отправитель закроет in после последней строки; тогда нужно закрыть подтверждения. Функция - единственный отправитель в acknowledgements, другой участник получает подтверждения параллельно.
func ConsumeWithAck(in <-chan string, acknowledgements chan<- int, apply func(string))

54. Нужно отправлять строки в out по одной, каждый раз ожидая их индекс из acknowledgements. Неверный индекс нужно запомнить как ошибку, но завершить обмен; после всех строк нужно закрыть out и дочитать подтверждения. Лишнее подтверждение тоже дает ошибку, иначе нужно вернуть nil. Другой участник отвечает один раз на строку, затем после закрытия out может прислать лишние подтверждения и закрывает свой канал. out открыт; функция - единственный отправитель и владелец его закрытия. lines менять нельзя.
func SendConfirmed(lines []string, out chan<- string, acknowledgements <-chan int) error

55. В одной фоновой горутине последовательно нужно вызывать call для values, обмениваясь запросами и ответами через два небуферизированных канала. Ошибка тоже занимает позицию в результате и не отменяет следующие вызовы; при ней Value = 0. Нужно вернуть ответы по порядку после завершения обработчика; values нельзя менять.
type Outcome struct { Value int; Err error }
func CallSerially(values []int, call func(int) (int, error)) []Outcome

56. В фоне нужно вызвать edit с counter и вернуть тот же указатель, не копируя структуру. Сразу нужно вернуть канал одного ответа; до ответа вызывающая сторона не обращается к counter, а после отправки указателя обработчик больше его не использует. counter не nil, edit не сохраняет указатель; нужно закрыть выход без ожидания читателя.
type Counter struct { Value int }
func StartOwnedEdit(counter *Counter, edit func(*Counter)) <-chan *Counter
